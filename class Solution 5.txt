class Solution {
    public int[] replaceElements(int[] arr) {
     for(int i=0; i<arr.length; i++){
       arr[i]=max(arr,i+1);
     }
     return arr;   
    }
    public static int max(int[] arr,int i){
        if(i == arr.length)return -1;

        int max = 0;
        for(int j=i; j<arr.length; j++){
            if(max<arr[j]){
                max = arr[j];
            }
        }
        return max;
    }

}
